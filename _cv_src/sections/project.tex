\section{Selected Open-Source Projects}

\subsection{Agent Infra}
\begin{enumerate}[label={[\arabic*]}]
\item \textbf{PenguinHarness}
\url{https://github.com/Prism-Shadow/penguin-harness} (\textbf{2.1k stars})
\begin{itemize}
\item Role: Chief Architect.
\item A local-first platform where agents autonomously build and improve other agents across the full app lifecycle, at far lower cost.
\end{itemize}

\item \textbf{GDPevo}
\url{https://prism-shadow.github.io/GDPevo/} (\textbf{62 stars})
\begin{itemize}
\item Role: Project Lead.
\item The first benchmark for agent self-evolution on economically valuable (GDP-related) tasks, lifting test accuracy by 17--22\% while cutting token cost.
\end{itemize}

\item \textbf{Awesome-RSI}
\url{https://github.com/Prism-Shadow/Awesome-RSI} (\textbf{55 stars})
\begin{itemize}
\item Role: Project Lead.
\item A curated hub of benchmarks and methods for Recursive Self-Improvement (RSI)---agents that improve themselves.
\end{itemize}
\end{enumerate}

\subsection{LLM Infra}
\begin{enumerate}[label={[\arabic*]}]
\item \textbf{RedKnot}
\url{https://github.com/rednote-machine-learning/RedKnot} (\textbf{1.7k stars})
\begin{itemize}
\item Role: Project Lead.
\item A long-context LLM inference engine on SGLang that cuts prefill FLOPs by 50\%--70\% for 1.35x--2.2x faster TTFT at near-lossless accuracy.
\end{itemize}
\end{enumerate}
